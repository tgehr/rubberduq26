QFT in Silq

def QFT[n:!ℕ](x: uint[n])mfree: uint[n]{
  for k in [0..n div 2){
    (x[k],x[n-k-1]) := (x[n-k-1],x[k]);
  }
  for k in [0..n){
    x[k] := H(x[k]);
    for l in [k+1..n){
      if x[k] && x[l]{
        phase(2·π·2^(k-l-1)); // 0-qubit gate
      }
    }
  }
  return x;
}

“Textbook” variant of Shor’s algorithm (for semiprimes)

import qft;

def main() {
  N := 3233; a := 10 coerce !ℤstar[n];
  n := 2·ceil(log₂(N))+1 coerce !ℕ;
  while true {
    t:=[H(0) for i in 0..n] as uint[n];
    measure(a^t);
    QFT(t) := t; // adjoint calls
    θ := (measure(t) as !ℕ)/2^n;
    for r in candidates(θ) {
      k := gcd(a^(r div 2)+1 as !ℕ,N);
      if 1<k && k<N { return k; }
    }
  }
}
def candidates(q:!ℚ): !ℕ[] {
  r := [];
  (b, d) := (1, 0);
  while q≠0 {
    q = 1/q;
    k := floor(q) coerce !ℕ;
    (b, d) = (k·b+d, b);
    if b>1 { r ~= [b]; }
    q -= k;
  }
  return r;
}

Iterative variant of Shor’s algorithm (for semiprimes; no explicit QFT)

def main() {
  n := 3233; a := 10 coerce !ℤstar[n];
  while true {
    (q, x) := (0, 1:!ℤstar[n]);
    for i in (2·ceil(log2(n))+1..-1..0] {
      bit := H(0);
      if bit {
        x ·= a^(2^i);
        phase(-2·π·q·2^i);
      }
      q += measure(H(bit))/2^(i+1);
    }
    measure(x);
    for r in candidates(q) {
      k := gcd(a^(r div 2)+1 as !ℕ, n);
      if 1<k && k<n { return k; }
    }
  }
}
def candidates(q:!ℚ): !ℕ[] {
  r := [];
  (b, d) := (1, 0);
  while q≠0 {
    q = 1/q;
    k := floor(q) coerce !N;
    (b, d) = (k·b+d, b);
    if b>1 { r ~= [b]; }
    q -= k;
  }
  return r;
}

Amplitude Amplification

def amplify_amplitude[n:!ℕ](
  const p: uint[n] !→lifted 𝔹,
  x: uint[n]
): uint[n] {
  if p(x) { phase(π); }

  with H_n(x) := x {
    if x ≠ 0 {
      phase(π);
    }
  }
  return x;
}
def H_n[n:!ℕ](x: uint[n]){
  for i in 0..n{ x[i] := H(x[i]); }
  return x;
}

def main() {
  def p(const x: uint[6]) ⇒ x = 42;
  x := H_n(0: uint[6]);
  repeat 6 {
    x := amplify_amplitude(p, x);
  }
  return x;
}

GQSP: Apply a bounded polynomial to a unitary.

def gqsp[d:!N](const P:!ℝ^(d+1))[a:qtype](
  const f: a ->mfree a
)(x: a): 𝔹 x a {
  (ts, ps, l) := compute_angles(P);
  anc := R(ts[0], ps[0], l)(0);
  for j in 1..d+1 {
    if anc = 0 { x := f(x); } // controlled
    anc := R(ts[j], ps[j], 0)(anc);
  }
  return (anc, x);
}

namespace Solution {
	open Microsoft.Quantum.Primitive;
	open Microsoft.Quantum.Canon;
    operation EvaluatePeriodClauses (queryRegister : Qubit[], periodAncillas : Qubit[]) : Unit {
        body (...) {
            let N = Length(queryRegister);
            for (period in 1 .. Length(periodAncillas)) {
                for (i in 0..N-period-1) {
                    CNOT(queryRegister[i + period], queryRegister[i]);
                }
                (ControlledOnInt(0, X))(queryRegister[0..N-period-1], periodAncillas[period-1]);
                for (i in N-period-1..-1..0) {
                    CNOT(queryRegister[i + period], queryRegister[i]);
                }
            }
        }
        adjoint auto;
    }
    operation Solve (x : Qubit[], y : Qubit) : Unit {
        body (...) { // Winter 2019 Q# coding contest master solution for problem C2
            let N = Length(x);
            using (anc = Qubit[N - 1]) {
                EvaluatePeriodClauses(x, anc);
                (ControlledOnInt(0, X))(anc, y);
                X(y);
                Adjoint EvaluatePeriodClauses(x, anc);
            }
        }
        adjoint self;
    }
}

Automatic uncomputation

  • Static analysis detects quantum states of the form ∣φ,ψ⟩=∑x​∣x,f(x)⟩.
  • For such states, coherent dropping is realizable: ∣φ,ψ⟩↦∣φ⟩.
  • Each expression is either used as a control (const) or consumed (moved).
  • Silq coherently drops quantum information that is not consumed (when possible).
  • Type checker catches programming mistakes.

Effects in Silq

Function annotations (can be inferred):

  • qfree: ∣x⟩↦∣f(x)⟩. f reversible classical function. Sometimes automatically uncomputable.
  • mfree: no measurements. Can control and take adjoint.
  • wild: general (partial) channel

Parameter annotations:

  • moved: consumed linearly
  • const: used as a control, remains caller’s responsibility

Example: Bitcoin EC Point Addition

  • lifted: sugar for annotating all parameters const, with qfree effect
p := 2^256-2^32-977 coerce !N;
a := 0, b := 7; // y^2 = x^3 + 7 (mod p)
Field := ℤmod[p];
Group := ℤstar[p];
APoint := 𝔹 x Field x Field; // (infinity, x, y)

def affine_infinity()lifted: APoint ⇒ (true,0,0): APoint;
def affine_neg(P: APoint)lifted: APoint{
	(infinity,x,y) := P;
	return (infinity,x,-y);
}
def affine_add(P: APoint, Q: APoint)lifted: APoint{
	(infinity1,x1,y1) := P;
	(infinity2,x2,y2) := Q;
	if infinity1{ return Q; }
	if infinity2{ return P; }
	if x1=x2{
		if y1≠y2 || y1=0{ return affine_infinity(); }
		λ := 3·x1·x1/(2·y1 pun Group);
		x3 := λ·λ-2·x1;
		y3 := λ·(x1-x3)-y1;
		return (false,x3,y3): APoint;
	}
	λ := (y2-y1)/(x2-x1 pun Group);
	x3 := λ·λ-x1-x2;
	y3 := λ·(x1-x3)-y1;
	return (false,x3,y3): APoint;
}
def affine_add_in_place(P: APoint, const Q: APoint)qfree: APoint{
	nP := affine_add(P, Q);
	affine_add(nP, affine_neg(Q)) := P;
	return nP;
}

Static analysis for automatic uncomputation (sketch)

  • For each variable y track dep(y): either TOP or a set of live variables
  • If dep(y)≠TOP, then it is automatically uncomputable
  • dep(y)={x₁,x₂,,…,xₙ} means that ∃f.⟨x,y,…∣φ⟩=0⇔y=f(x1​,x2​,…,xn​).
  • When a variable gets consumed, substitute its dependency set into dependents or propagate TOP.

Uncomputation and Adjoint Synthesis Example

Recall we are working with Turing-complete, recursive programs.

def rec[n:!ℕ][m:!ℕ](const x:𝔹^m, y:𝔹^m)qfree:𝔹^m{
	if n ≠ 0{
		t := vector(m, 1:𝔹);
		t := rec[n sub 1](x, t);
		y := f(t, y);
		forget(t); // (explicit for emphasis)
	}else{
		y := f(x, y);
	}
	return y;
}

Naive Uncomputation Synthesis

def rec[n:!ℕ][m:!ℕ](const x:𝔹^m, y:𝔹^m)qfree:𝔹^m{
	if n ≠ 0{
		t := vector(m, 1:𝔹);
		t := rec[n sub 1](x, t);
		y := f(t, y);
		rec[n sub 1](x,t) := t; // forget(t)
		vector(m,1:𝔹) := t;
	}else{
		y := f(x, y);
	}
	return y;
}

Keeping around the garbage

(Pseudocode)

def rec_g[n:!ℕ][m:!ℕ](const x:𝔹^m, y:𝔹^m, ref g:Garbage)qfree:B^m{
	if n ≠ 0{
		t := vector(m, 1:𝔹);
		t := rec_g[n sub 1](x, t, g);
		y := f(t, y);
		g.dispose(t);
	}else{
		y := f(x, y);
	}
	return y;
}

Garbage-consuming adjoint

(Pseudocode)

def rec_g_adj[n:!ℕ][m:!ℕ](const x:𝔹^m, y:𝔹^m, ref g:Garbage)qfree:𝔹^m{
	if n ≠ 0{
		t := g.pick[𝔹^m]();
		f(t,y) := y;
		t := rec_g_adj[n sub 1](x, t, g);
		vector(m, 1:𝔹) := t;
	}else{
		f(x,y) := y;
	}
	return y;
}

rec with efficient uncomputation

def rec[n:!ℕ][m:!ℕ](const x:𝔹^m, y:𝔹^m)qfree:𝔹^m{
	if n != 0{
		t := vector(m,1:𝔹);
		g := empty_garbage();
		t := rec_g[n sub 1](x, t, g);
		y := f(t,y);

		empty_garbage() := g;

	}else{

	}
	return y;
}

Quantum programming with isometries

  • Unitary operators require output qubits to be allocated up-front and all input qubits have to be returned.
def H_n[n:!ℕ](qs:𝔹^n):𝔹^n{ // unitary, n-qubit Hadamard transform
	for i in [0..n){
		qs[i] := H(qs[i]);
	}
	return qs;
}
  • Isometries can allocate qubits internally:
def entangle_H[n:!ℕ](q:𝔹):𝔹^n{ // isometry
	assert(n ≥ 1);
	return H_n([dup(q) for i in [0..n-1)]~[q]);
}

Partial Isometries (Unsafe)

  • Partial isometries can deallocate qubits in known states.
def dealloc_plus(x:𝔹){ // partial isometry from 𝔹 to 𝟙
	H(0) := x; // adjoint of x := H(0)
}
  • This is unsafe, there is no deterministic way to check validity on a quantum computer.
  • However, the intent is made clear and therefore errors can be caught in a simulator.
  • Can be interpreted e.g. as assert(measure(H(x))=0) or as trusted reset-to-zero
  • Raises interesting questions about semantics.

Control and Scoping

  • There is both classical if and quantum if (ad-hoc polymorphism).
if measure(H(0)) { ... }

x := H(0);
if x{ ... } // note: `if H(0){ ...}` not valid
  • Variables can be declared conditionally in branches and escape them:
x := H(0);
if x{
	t := H(0);
	y := t;
}else{
	y := 1;
}
return (x,y);
  • Key benefit: This does not force a specific register allocation.
